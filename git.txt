---------УСТАНОВКА И НАСТРОЙКА---------
git --version 				# проверка версии
git config --global user.name "ИМЯ" 	# имя того, кто делает коммиты
git config --global user.email "АДРЕС" 	# почта того, кто делает коммиты
git config --global --list 		# показать все сохрранённые настройки
git config --global init.defaultBranch ИМЯ # настройка название основной ветки (обычно main)

----------СОЗДАНИЕ РЕПОЗИТОРИЯ----------
git init 	# инициализация репозитория
git status 	# проверка статуса репозитория

----------СОХРАНЕНИЕ В ИСТОРИИ----------
git add ФАЙЛ 		# подготовка файла к сохранению
git commit -m ФАЙЛ 	# сохранить подготовленные изменения в историю

-------------ПРОСМОТР ИСТОРИИ------------
git log 		# показать историю всех коммитов в текущей ветке
git log --oneline 	# показать историю в очень сжатом виде
git diff 		# показать разницу между тем, что сейчас в файлах, и тем, что было в последнем коммите 
---------ОТКАТ ИЗМЕНЕНИЙ---------- 
git restore --staged ФАЙЛ 	# убрать файл из зелёной зоны (подготовка к сохранению) в рабочую область 
				  (сделать его красным). Изменения в файле сохранятся
git restore ФАЙЛ 		# вернуть файл к тому виду, в котором он был в последнем коммите.
				  ВНИМАНИЕ: Все несохранённые изменения в этом файле будут удалены
----------СОЗДАНИЕ ВЕТКИ----------
git branch # показать список всех веток в проекте
git branch ИМЯ # создать новую ветку
git switch ИМЯ # переключится на другую ветку
git switch -с ИМЯ # создать ветку и сразу переключится на неё
